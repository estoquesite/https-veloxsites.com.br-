Testei a demonstração do sistema de estoque e quero contratar a versão completa e ilimitada para a minha empresa.", "_blank"); return; // Interrompe o cadastro do 6º produto } // Captura os valores dos inputs do formulário // (Certifique-se de que os IDs no seu HTML correspondem a estes: 'nome-produto', 'qtd-produto', 'preco-produto') const nomeInput = document.getElementById('nome-produto'); const qtdInput = document.getElementById('qtd-produto'); const precoInput = document.getElementById('preco-produto'); if (!nomeInput || !qtdInput || nomeInput.value.trim() === '' || qtdInput.value.trim() === '') { alert("Por favor, preencha o nome e a quantidade do produto!"); return; } const novoProduto = { id: Date.now(), nome: nomeInput.value.trim(), quantidade: qtdInput.value.trim(), preco: precoInput ? precoInput.value.trim() : '0,00' }; produtos.push(novoProduto); salvarProdutos(produtos); // Limpa os campos após salvar nomeInput.value = ''; qtdInput.value = ''; if (precoInput) precoInput.value = ''; carregarProdutos(); alert("✅ Produto cadastrado com sucesso! (Demonstração: " + produtos.length + "/5 produtos utilizados)"); } // Carrega e exibe os produtos na tabela do site function carregarProdutos() { const produtos = obterProdutos(); const tabela = document.getElementById('tabela-produtos'); // ID da sua tabela no HTML if (!tabela) return; tabela.innerHTML = ''; produtos.forEach((prod) => { const linha = document.createElement('tr'); linha.innerHTML = ` ${prod.nome} ${prod.quantidade} R$ ${prod.preco} `; tabela.appendChild(linha); }); } // Remove um produto da lista function removerProduto(id) { let produtos = obterProdutos(); produtos = produtos.filter(prod => prod.id !== id); salvarProdutos(produtos); carregarProdutos(); }
